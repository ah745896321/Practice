public class Solution

    //Dictionary of all possible characters for every digit
    //Strings can be used inside instead of Lists but that's a formality
    public Dictionary<char, List<char>> map = new Dictionary<char, List<char>>()
    {
        {'2', new List<char>(){ 'a', 'b', 'c'}},
        {'3', new List<char>(){ 'd', 'e', 'f'}},
        {'4', new List<char>(){ 'g', 'h', 'i'}},
        {'5', new List<char>(){ 'j', 'k', 'l'}},
        {'6', new List<char>(){ 'm', 'n', 'o'}},

可用DFS  寫起來比較乾淨
    public List<string> output = new List<string>(){};


    public IList<string> LetterCombinations(string digits) 
    {
        //Check if the string is not empty
        if(String.IsNullOrEmpty(digits)){ return output; }

        //start with first digit
        dfs(digits, 0, "");

        return output;
    }

    //DFS method to go through all digits recursively in every combination
    public void dfs(string digits, int index, string word)
    {
        //Base condition if 1 digit = 1 character then length of word at the end has to be the same as input digits length
        if(index == digits.Length)
        {
            output.Add(word);
            return;
        }

        //Going through all letters a digit could be and passing down an appended string
        foreach(char c in map[digits[index]])
        {
            dfs(digits, index + 1, word + c);
        }
    }
}
