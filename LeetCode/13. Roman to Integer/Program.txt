public class Solution {
    public int RomanToInt(string s) {
        int result =0;
        int cur =0;
        int pre =0;
        foreach(char c in s)
        {
            switch (c)
            {
                case 'I':
                cur=1;
                break;
                case 'V':
                cur=5;
                break;
                case 'X':
                cur=10;
                break;
                case 'L':
                cur=50;
                break;
                case 'C':
                cur=100;
                break;
                case 'D':
                cur=500;
                break;
                case 'M':
                cur=1000;
                break;
            }
            if(pre<cur)
            result-=pre;
            else
            result+=pre;
            pre=cur;
        }
        result+=pre;
        return result;
    }
}
