public class Solution
{
    public bool IsValid(string s)
    {
        Stack<char> stack = new Stack<char>();
        char temp = s.First();
        if (s.Length % 2 == 1||temp == '}' || temp == ']' || temp == ')') return false;
        foreach (char c in s)
        {
            if (c == '('|| c == '['|| c == '{')
                stack.Push(c);
            else
            {
                if (stack.Count == 0)
                {
                    return false;
                }
                temp = stack.Pop();
                if (temp == '{' && c == '}' || temp == '[' && c == ']' || temp == '(' && c == ')')
                { }
                else
                    return false;
            }
            
            
            
        }
        return stack.Count == 0;
    }
}
